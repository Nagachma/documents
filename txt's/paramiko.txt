import paramiko

# Define the remote host and login credentials
hostname = '10.70.10.25'  # Ensure this is a valid IP or hostname
port = 22  # Ensure this is the correct port
username = 'orchestrai_read_user'
password = 'hK9!tZdMl'

# Define the shell script you want to execute
shell_script = """
#!/bin/bash
echo "This is a sample shell script execution"
date
uptime
"""

# Initialize the SSH client
ssh = paramiko.SSHClient()
ssh.set_missing_host_key_policy(paramiko.AutoAddPolicy())

try:
    # Connect to the remote host
    ssh.connect(hostname=hostname, port=port, username=username, password=password)

    # Execute the shell script
    stdin, stdout, stderr = ssh.exec_command(shell_script)

    # Print the outputs
    print("Standard Output:")
    for line in stdout.readlines():
        print(line, end="")

    print("\nStandard Error:")
    for line in stderr.readlines():
        print(line, end="")
finally:
    # Close the connection
    # sh.close()
    stdin.close()
    stdout.close()
    stderr.close()
    ssh.close()
